cmake_minimum_required(VERSION 3.20)
project(saelis_research_lab VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
add_library(saelis_research src/research.cpp)
add_library(SAELIS::Research ALIAS saelis_research)
target_include_directories(saelis_research PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_compile_features(saelis_research PUBLIC cxx_std_20)
if(BUILD_TESTING)
 add_executable(saelis_research_tests tests/research_tests.cpp)
 target_link_libraries(saelis_research_tests PRIVATE SAELIS::Research)
 add_test(NAME saelis.research COMMAND saelis_research_tests)
endif()
